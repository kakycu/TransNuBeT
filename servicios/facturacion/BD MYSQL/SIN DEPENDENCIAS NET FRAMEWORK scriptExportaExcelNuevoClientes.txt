Dim contadorFacturasGlobal As Integer

Sub ExportarTodasFacturasFormatoCompacto()
    Dim ws As Worksheet
    Dim libroOrigen As Workbook
    Dim rutaArchivo As String
    Dim contenidoTotal As String
    Dim contadorFacturas As Integer
    Dim fso As Object, archivo As Object
    Dim facturasData As Object ' Scripting.Dictionary
    Dim detallesData As Object ' Scripting.Dictionary
    Dim facturaIndex As Long
    Dim detalleIndex As Long
    
    Dim idInicialFact As Long
    Dim idInicialDet As Long
    Dim respuestaIdFact As String
    Dim respuestaIdDet As String
    Dim idActualFact As Long
    Dim idActualDet As Long
    
    Set libroOrigen = ThisWorkbook
    contadorFacturas = 0
    contenidoTotal = ""
    
    ' Usar Dictionary en lugar de ArrayList
    Set facturasData = CreateObject("Scripting.Dictionary")
    Set detallesData = CreateObject("Scripting.Dictionary")
    
    facturaIndex = 0
    detalleIndex = 0
    
    ' Inicializar contador global de números de factura
    contadorFacturasGlobal = 1
    
    ' Solicitar archivo destino
    rutaArchivo = Application.GetSaveAsFilename( _
        InitialFileName:="Facturas_Compacto.txt", _
        FileFilter:="Archivos de texto (*.txt), *.txt", _
        Title:="Guardar archivo de texto")
    
    If rutaArchivo = "False" Then Exit Sub
    
    ' Preguntar por el ID inicial para tbl_fact
    respuestaIdFact = InputBox("¿En qué número debe comenzar el ID de la tabla 'tbl_fact'?", _
                               "Configuración SQL - Facturas", "1")
                           
    ' Validar que sea un número, si no, usar 1 por defecto
    If IsNumeric(respuestaIdFact) And respuestaIdFact <> "" Then
        idInicialFact = CLng(respuestaIdFact)
    Else
        idInicialFact = 1
    End If
    
    ' Preguntar por el ID inicial para tbl_fact_detalle
    respuestaIdDet = InputBox("¿En qué número debe comenzar el ID de la tabla 'tbl_fact_detalle'? (Se incrementará automáticamente para cada detalle)", _
                              "Configuración SQL - Detalles", "1")
                           
    ' Validar que sea un número, si no, usar 1 por defecto
    If IsNumeric(respuestaIdDet) And respuestaIdDet <> "" Then
        idInicialDet = CLng(respuestaIdDet)
    Else
        idInicialDet = 1
    End If
    
    ' Inicializar IDs actuales
    idActualFact = idInicialFact
    idActualDet = idInicialDet
    
    ' Procesar cada hoja (excepto las que tienen nombres específicos)
    For Each ws In libroOrigen.Worksheets
        ' Saltar hojas con nombres específicos
        If LCase(ws.Name) = "instrucciones" Or _
           LCase(ws.Name) = "resumen-enero" Or _
           LCase(ws.Name) = "resumen" Or _
           LCase(Left(ws.Name, 7)) = "resumen" Then
            ' No hacer nada, saltar esta hoja
        ElseIf ws.Visible Then
            ' Procesar la hoja pasando los IDs actuales
            Dim contenidoHoja As String
            Dim idFactUsado As Long
            contenidoHoja = ExtraerContenidoCompacto(ws, facturasData, detallesData, facturaIndex, detalleIndex, _
                                                     idActualFact, idActualDet, idFactUsado)
            
            If contenidoHoja <> "" Then
                contenidoTotal = contenidoTotal & contenidoHoja
                contadorFacturas = contadorFacturas + 1
                ' Actualizar ID actual para la siguiente factura (incrementar en 1)
                idActualFact = idFactUsado + 1
                ' NOTA: idActualDet se actualiza dentro de ExtraerContenidoCompacto
            End If
        End If
    Next ws

    ' Agregar estructura de base de datos y INSERTs SQL al final
    contenidoTotal = contenidoTotal & GenerarEstructuraSQL(facturasData, detallesData, idInicialFact, idInicialDet)
    
    ' Crear y escribir archivo
    Set fso = CreateObject("Scripting.FileSystemObject")
    Set archivo = fso.CreateTextFile(rutaArchivo, True, True)
    archivo.Write contenidoTotal
    archivo.Close
    
    ' Abrir el archivo generado
    Shell "notepad.exe " & Chr(34) & rutaArchivo & Chr(34), vbNormalFocus
    
    MsgBox "Exportación completada:" & vbCrLf & _
           "Facturas procesadas: " & contadorFacturas & vbCrLf & _
           "ID inicial facturas: " & idInicialFact & vbCrLf & _
           "ID inicial detalles: " & idInicialDet & vbCrLf & _
           "Último ID de detalle usado: " & (idActualDet - 1) & vbCrLf & _
           "Archivo: " & rutaArchivo, vbInformation
    
    ' Limpiar
    Set fso = Nothing
    Set archivo = Nothing
    Set libroOrigen = Nothing
    Set facturasData = Nothing
    Set detallesData = Nothing
End Sub

Function ExtraerContenidoCompacto(ws As Worksheet, ByRef facturasData As Object, ByRef detallesData As Object, _
                                  ByRef facturaIndex As Long, ByRef detalleIndex As Long, _
                                  ByVal idActualFact As Long, ByRef idActualDet As Long, _
                                  ByRef idFactUsado As Long) As String
    Dim contenido As String
    Dim numFactura As String, cliente As String, fechaEmision As String
    Dim observaciones As String
    Dim totalGeneral As Double
    Dim i As Integer, startRow As Integer
    Dim descripcion As String, cantidad As String, precio As String, importe As String
    Dim productCount As Integer
    Dim facturaId As Long
    Dim j As Integer
    Dim posicionDetalle As Integer
    Dim idDetalleActual As Long
    
    ' Extraer No. Factura
    numFactura = ExtraerNumeroFacturaSimple(ws)
    
    ' Extraer Cliente
    cliente = ExtraerClienteLimpio(ws)
    
    ' Extraer Fecha Emisión
    fechaEmision = ExtraerFechaSola(ws)
    
    ' Extraer observaciones completas
    observaciones = ExtraerObservacionesCompletas(ws)
    
    ' Procesar productos primero para obtener el total
    startRow = 1
    For i = 1 To 20
        If Not IsError(ws.Cells(i, 2).Value) Then
            Dim cellValue As String
            cellValue = CStr(ws.Cells(i, 2).Value)
            
            If cellValue = "Descripcion:" Or cellValue = "Descipcion:" Then
                startRow = i + 1
                Exit For
            End If
        End If
    Next i
    
    ' Si no encuentra, empezar en fila 8
    If startRow = 1 Then startRow = 8
    
    ' Procesar productos para calcular total
    i = startRow
    productCount = 0
    totalGeneral = 0
    
    Do While i <= 100
        ' Obtener valores con limpieza completa
        On Error Resume Next
        descripcion = LimpiarTexto(ws.Cells(i, 2).Value)
        cantidad = LimpiarTexto(ws.Cells(i, 4).Value)
        precio = LimpiarTexto(ws.Cells(i, 5).Value)
        importe = LimpiarTexto(ws.Cells(i, 6).Value)
        On Error GoTo 0
        
        ' Si la descripción está vacía, salir
        If descripcion = "" Or descripcion = "Descripcion:" Or descripcion = "Descipcion:" Then
            Exit Do
        End If
        
        ' Obtener código basado en la descripción
        Dim codigoServicio As String
        codigoServicio = ObtenerCodigoServicioSimple(descripcion)
        
        ' Calcular valores reales
        Dim cantidadNum As Double, precioNum As Double, importeNum As Double
        cantidadNum = ConvertirCantidadSimple(cantidad)
        precioNum = ConvertirPrecioSimple(precio)
        
        ' Calcular importe si es fórmula o vacío
        If IsNumeric(importe) Then
            importeNum = CDbl(importe)
        Else
            importeNum = cantidadNum * precioNum
        End If
        
        totalGeneral = totalGeneral + importeNum
        
        productCount = productCount + 1
        i = i + 1
    Loop
    
    ' Buscar Total General en F14-F16 (si no se encontró en productos)
    If totalGeneral = 0 Then
        totalGeneral = BuscarTotalGeneralSimple(ws)
    End If
    
    ' Línea de encabezado
    contenido = numFactura & "," & cliente & "," & fechaEmision & "," & Format(totalGeneral, "0.00") & vbCrLf
    
    ' Ahora procesar productos para mostrar en el contenido
    i = startRow
    Do While i <= 100
        ' Obtener valores con limpieza completa
        On Error Resume Next
        descripcion = LimpiarTexto(ws.Cells(i, 2).Value)
        cantidad = LimpiarTexto(ws.Cells(i, 4).Value)
        precio = LimpiarTexto(ws.Cells(i, 5).Value)
        importe = LimpiarTexto(ws.Cells(i, 6).Value)
        On Error GoTo 0
        
        ' Si la descripción está vacía, salir
        If descripcion = "" Or descripcion = "Descripcion:" Or descripcion = "Descipcion:" Then
            Exit Do
        End If
        
        ' Obtener código basado en la descripción
        codigoServicio = ObtenerCodigoServicioSimple(descripcion)
        
        ' Calcular valores reales
        cantidadNum = ConvertirCantidadSimple(cantidad)
        precioNum = ConvertirPrecioSimple(precio)
        
        ' Calcular importe si es fórmula o vacío
        If IsNumeric(importe) Then
            importeNum = CDbl(importe)
        Else
            importeNum = cantidadNum * precioNum
        End If
        
        ' Formatear línea del producto
        contenido = contenido & codigoServicio & "," & _
                    descripcion & "," & _
                    Format(cantidadNum, "0") & "," & _
                    Format(importeNum, "#,##0.00") & ", Precio: " & _
                    Format(precioNum, "0.00") & vbCrLf
        
        i = i + 1
    Loop
    
    ' Si no hay productos, agregar mensaje
    If productCount = 0 Then
        contenido = contenido & "SIN PRODUCTOS" & vbCrLf
    End If
    
    ' Agregar total
    contenido = contenido & "Total Factura No." & numFactura & ": " & Format(totalGeneral, "#,##0.00") & vbCrLf
    
    ' Agregar observaciones completas
    contenido = contenido & "Observaciones: " & observaciones & vbCrLf
    
    ' Línea separadora
    contenido = contenido & String(80, "=") & vbCrLf & vbCrLf
    
    ' Almacenar datos para SQL
    If numFactura <> "" And totalGeneral > 0 Then
        ' Asignar el ID actual a esta factura
        facturaId = idActualFact
        
        ' Guardar factura en Dictionary
        Dim facturaKey As String
        facturaKey = "F" & facturaIndex
        facturaIndex = facturaIndex + 1
        
        facturasData.Add facturaKey, Array( _
            facturaId, _
            numFactura, _
            cliente, _
            fechaEmision, _
            totalGeneral, _
            observaciones, _
            productCount)
        
        ' Guardar detalles - CADA DETALLE USA UN ID SECUENCIAL ÚNICO
        i = startRow
        posicionDetalle = 0
        
        Do While i <= 100
            On Error Resume Next
            descripcion = LimpiarTexto(ws.Cells(i, 2).Value)
            cantidad = LimpiarTexto(ws.Cells(i, 4).Value)
            precio = LimpiarTexto(ws.Cells(i, 5).Value)
            importe = LimpiarTexto(ws.Cells(i, 6).Value)
            On Error GoTo 0
            
            If descripcion = "" Or descripcion = "Descripcion:" Or descripcion = "Descipcion:" Then
                Exit Do
            End If
            
            codigoServicio = ObtenerCodigoServicioSimple(descripcion)
            cantidadNum = ConvertirCantidadSimple(cantidad)
            precioNum = ConvertirPrecioSimple(precio)
            
            If IsNumeric(importe) Then
                importeNum = CDbl(importe)
            Else
                importeNum = cantidadNum * precioNum
            End If
            
            ' Guardar detalle - CADA DETALLE TIENE UN ID ÚNICO Y SECUENCIAL
            Dim detalleKey As String
            detalleKey = "D" & detalleIndex
            detalleIndex = detalleIndex + 1
            
            ' Guardamos: facturaId, código, descripción, cantidad, precio, importe, idDetalle
            detallesData.Add detalleKey, Array( _
                facturaId, _
                codigoServicio, _
                descripcion, _
                cantidadNum, _
                precioNum, _
                importeNum, _
                idActualDet)  ' ID SECUENCIAL ÚNICO para este detalle
            
            i = i + 1
            posicionDetalle = posicionDetalle + 1
            idActualDet = idActualDet + 1  ' INCREMENTAR SIEMPRE EN 1 PARA CADA DETALLE
        Loop
        
        ' Devolver el ID usado para esta factura
        idFactUsado = facturaId
    Else
        idFactUsado = idActualFact - 1
    End If
    
    ExtraerContenidoCompacto = contenido
End Function
Function GenerarEstructuraSQL(facturasData As Object, detallesData As Object, ByVal idInicialFact As Long, ByVal idInicialDet As Long) As String
    Dim contenidoSQL As String
    Dim i As Long
    Dim valoresFacturas As String
    Dim valoresDetalles As String
    Dim facturaKey As Variant
    Dim detalleKey As Variant
    Dim facturaArray As Variant
    Dim detalleArray As Variant
    
    contenidoSQL = contenidoSQL & "--" & vbCrLf
    contenidoSQL = contenidoSQL & "-- Volcado de datos para la tabla `tbl_fact`" & vbCrLf
    contenidoSQL = contenidoSQL & "-- (IDs comenzando desde: " & idInicialFact & ")" & vbCrLf
    contenidoSQL = contenidoSQL & "--" & vbCrLf & vbCrLf
    
    ' Generar INSERTs para tbl_fact
    valoresFacturas = ""
    i = 0
    
    For Each facturaKey In facturasData.Keys
        If valoresFacturas <> "" Then valoresFacturas = valoresFacturas & "," & vbCrLf
        
        facturaArray = facturasData(facturaKey)
        
        ' EXTRAER VALORES EN VARIABLES INDIVIDUALES
        Dim facturaIdVal As Variant
        Dim noFacturaVal As String
        Dim clienteVal As String
        Dim fechaVal As String
        Dim totalVal As Double
        Dim observacionesVal As String
        
        facturaIdVal = facturaArray(0)
        noFacturaVal = facturaArray(1)
        clienteVal = facturaArray(2)
        fechaVal = facturaArray(3)
        totalVal = facturaArray(4)
        observacionesVal = facturaArray(5)
        
        ' Obtener cliente_id
        Dim clienteId As Integer
        clienteId = ObtenerClienteIdDeClasifClientes(clienteVal)
        
        ' CORREGIDO: fecha_pago = NULL, Ref_pago = NULL (sin coma extra)
        valoresFacturas = valoresFacturas & "(" & facturaIdVal & ", '" & EscapeSQL(noFacturaVal) & "', " & _
                         clienteId & ", 20, " & _
                         Format(CDbl(totalVal), "0.00") & ", " & _
                         Format(CDbl(totalVal), "0.00") & ", 1, 'CONTABILIZADA', " & _
                         "'" & ConvertirFechaMySQL(fechaVal) & "', " & _
                         "'" & ConvertirFechaMySQL(fechaVal) & "', " & _
                         "NULL, NULL, " & _
                         "'" & EscapeSQL(observacionesVal) & "', 1)"
        
        i = i + 1
    Next facturaKey
    
    If valoresFacturas <> "" Then
        contenidoSQL = contenidoSQL & "INSERT INTO `tbl_fact` (`id`, `no_fact`, `cliente_id`, `tipo_pago_id`, `subtotal`, `total_general`, `usuario_id`, `estado`, `fecha_emision`, `fecha_contabilizacion`, `fecha_pago`, `Ref_pago`, `observaciones`, `usuario_contabilizacion`) VALUES" & vbCrLf
        contenidoSQL = contenidoSQL & valoresFacturas & ";" & vbCrLf & vbCrLf & vbCrLf
    End If

    contenidoSQL = contenidoSQL & "--" & vbCrLf
    contenidoSQL = contenidoSQL & "-- Volcado de datos para la tabla `tbl_fact_detalle`" & vbCrLf
    contenidoSQL = contenidoSQL & "-- (IDs comenzando desde: " & idInicialDet & " y aumentando secuencialmente)" & vbCrLf
    contenidoSQL = contenidoSQL & "--" & vbCrLf & vbCrLf
    
    ' Generar INSERTs para tbl_fact_detalle
    valoresDetalles = ""
    
    For Each detalleKey In detallesData.Keys
        If valoresDetalles <> "" Then valoresDetalles = valoresDetalles & "," & vbCrLf
        
        detalleArray = detallesData(detalleKey)
        
        ' EXTRAER VALORES
        Dim facturaIdDet As Variant
        Dim codigoServicioDet As String
        Dim cantidadDet As Double
        Dim precioDet As Double
        Dim importeDet As Double
        Dim idDetalle As Long
        
        facturaIdDet = detalleArray(0)
        codigoServicioDet = detalleArray(1)
        ' descripcionDet = detalleArray(2) ' No se usa
        cantidadDet = detalleArray(3)
        precioDet = detalleArray(4)
        importeDet = detalleArray(5)
        idDetalle = detalleArray(6)  ' ID SECUENCIAL ÚNICO para este detalle
        
        ' Obtener servicio_id
        Dim servicioId As Integer
        servicioId = ObtenerServicioIdSQL(codigoServicioDet)
        
        valoresDetalles = valoresDetalles & "(" & idDetalle & ", " & facturaIdDet & ", " & _
                         servicioId & ", " & _
                         Format(CDbl(cantidadDet), "0") & ", " & _
                         Format(CDbl(precioDet), "0.00") & ", " & _
                         Format(CDbl(importeDet), "0.00") & ")"
        
    Next detalleKey
    
    If valoresDetalles <> "" Then
        contenidoSQL = contenidoSQL & "INSERT INTO `tbl_fact_detalle` (`id`, `factura_id`, `servicio_id`, `cantidad`, `precio_unitario`, `total_linea`) VALUES" & vbCrLf
        contenidoSQL = contenidoSQL & valoresDetalles & ";" & vbCrLf
    End If
    
    GenerarEstructuraSQL = contenidoSQL
End Function

Function ObtenerClienteIdDeClasifClientes(ByVal nombreCliente As String) As Integer

    Dim nombreUpper As String
    nombreUpper = UCase(Trim(nombreCliente))
    
    ' Buscar coincidencias parciales priorizando las más específicas
    Select Case True
        
        ' SERVICIOS DE SALUD Y HOSPITALES
        Case InStr(nombreUpper, "HOSPITAL") > 0 And InStr(nombreUpper, "MARTIN CHANG") > 0
            ObtenerClienteIdDeClasifClientes = 98 ' UPR Hospital General Docente Martín Chang Puga
        Case InStr(nombreUpper, "SALUD PÚBLICA") > 0 Or InStr(nombreUpper, "SALUD PUBLICA") > 0
            ObtenerClienteIdDeClasifClientes = 2 ' UPR Dirección Municipal de Salud Pública Nuevitas

            ObtenerClienteIdDeClasifClientes = 24 ' Universidad de Ciencias Médicas

            ObtenerClienteIdDeClasifClientes = 14 ' Unidad de Aseguramiento y Apoyo a la salud

Function ConvertirFechaMySQL(ByVal fechaTexto As String) As String
    ' Convertir fecha de formato dd/mm/yyyy a YYYY-MM-DD HH:MM:SS
    Dim fechaParts() As String
    Dim dia As String, mes As String, anio As String
    
    fechaTexto = Trim(fechaTexto)
    
    If fechaTexto Like "##/##/####" Or fechaTexto Like "##/##/##" Then
        fechaParts = Split(fechaTexto, "/")
        If UBound(fechaParts) = 2 Then
            dia = Format(fechaParts(0), "00")
            mes = Format(fechaParts(1), "00")
            anio = fechaParts(2)
            If Len(anio) = 2 Then anio = "20" & anio
            
            ConvertirFechaMySQL = anio & "-" & mes & "-" & dia & " 08:00:00"
            Exit Function
        End If
    End If
    
    ConvertirFechaMySQL = "2025-01-06 08:00:00"
End Function

Function EscapeSQL(ByVal texto As String) As String
    ' Función para escapar caracteres especiales en SQL
    texto = LimpiarTexto(texto)
    
    If IsNull(texto) Or texto = "" Then
        EscapeSQL = ""
        Exit Function
    End If
    
    Dim result As String
    result = CStr(texto)
    result = Replace(result, "'", "''")
    result = Replace(result, "\", "\\")
    result = Trim(result)
    
    ' Limitar longitud para observaciones
    If Len(result) > 200 Then
        result = Left(result, 197) & "..."
    End If
    
    EscapeSQL = result
End Function

Function ExtraerClienteLimpio(ws As Worksheet) As String
    ' Extraer cliente de celda B6 y eliminar "Cliente:" si está presente
    Dim clienteCompleto As String
    clienteCompleto = ""
    
    On Error Resume Next
    clienteCompleto = LimpiarTexto(ws.Cells(6, 2).Value)
    On Error GoTo 0
    
    If clienteCompleto = "" Then
        ExtraerClienteLimpio = ws.Name
        Exit Function
    End If
    
    clienteCompleto = Replace(clienteCompleto, "Cliente:", "")
    clienteCompleto = Replace(clienteCompleto, "CLIENTE:", "")
    clienteCompleto = Replace(clienteCompleto, "cliente:", "")
    clienteCompleto = Replace(clienteCompleto, "Cliente", "")

Function ExtraerFechaSola(ws As Worksheet) As String
    ' Extraer solo la fecha de la celda D3, sin la palabra "Fecha"
    Dim fechaCompleta As String
    fechaCompleta = ""
    
    On Error Resume Next
    fechaCompleta = Trim(CStr(ws.Cells(3, 4).Value))
    On Error GoTo 0
    
    If fechaCompleta = "" Then
        ExtraerFechaSola = "Sin fecha"
        Exit Function
    End If
    
    fechaCompleta = Replace(fechaCompleta, "Fecha", "")
    fechaCompleta = Replace(fechaCompleta, "FECHA", "")
    fechaCompleta = Replace(fechaCompleta, "fecha", "")
    fechaCompleta = Trim(fechaCompleta)
    
    If Left(fechaCompleta, 1) = ":" Then
        fechaCompleta = Mid(fechaCompleta, 2)
        fechaCompleta = Trim(fechaCompleta)
    End If
    
    If InStr(1, fechaCompleta, "/") > 0 Then
        Dim fechaParts() As String
        fechaParts = Split(fechaCompleta, " ")
        If UBound(fechaParts) >= 0 Then
            If InStr(1, fechaParts(0), "/") > 0 Then
                ExtraerFechaSola = Trim(fechaParts(0))
                Exit Function
            End If
        End If
    End If
    
    ExtraerFechaSola = fechaCompleta
End Function

Function ExtraerNumeroFacturaSimple(ws As Worksheet) As String
    Dim i As Integer, j As Integer
    Dim fechaFactura As String
    Dim fechaYYYYMMDD As String
    Dim numeroEncontrado As String
    Dim regex As Object, matches As Object
    Dim celdaTexto As String
    
    fechaFactura = ExtraerFechaSola(ws)
    fechaYYYYMMDD = ConvertirFechaYYYYMMDD(fechaFactura)
    
    Set regex = CreateObject("VBScript.RegExp")
    regex.Pattern = "\d+"
    regex.Global = True
    
    numeroEncontrado = ""
    
    For i = 1 To 50
        For j = 4 To 8
            If Not IsError(ws.Cells(i, j).Value) Then
                celdaTexto = Trim(CStr(ws.Cells(i, j).Value))
                
                If InStr(1, UCase(celdaTexto), "FACTURA") > 0 Or _
                   InStr(1, UCase(celdaTexto), "NO.") > 0 Or _
                   InStr(1, UCase(celdaTexto), "FOLIO") > 0 Or _
                   InStr(1, UCase(celdaTexto), "N°") > 0 Then
                   
                    Set matches = regex.Execute(celdaTexto)
                    If matches.Count > 0 Then
                        numeroEncontrado = matches(matches.Count - 1).Value
                        GoTo ConstruirCodigo
                    End If
                    
                    Dim valorDerecha As String
                    If j < 10 Then
                        valorDerecha = Trim(CStr(ws.Cells(i, j + 1).Value))
                        Set matches = regex.Execute(valorDerecha)
                        If matches.Count > 0 Then
                            numeroEncontrado = matches(0).Value
                            GoTo ConstruirCodigo
                        End If
                    End If
                End If
                
                If InStr(1, UCase(celdaTexto), "FV-") > 0 Then
                    Set matches = regex.Execute(celdaTexto)
                    If matches.Count > 0 Then
                        numeroEncontrado = matches(matches.Count - 1).Value
                        GoTo ConstruirCodigo
                    End If
                End If
            End If
        Next j
    Next i
    
    If numeroEncontrado = "" Then
        On Error Resume Next
        celdaTexto = ws.Cells(16, 6).Value
        Set matches = regex.Execute(celdaTexto)
        If matches.Count > 0 Then
            numeroEncontrado = matches(0).Value
        End If
        On Error GoTo 0
    End If

ConstruirCodigo:
    If numeroEncontrado <> "" Then
        ExtraerNumeroFacturaSimple = "FV-" & fechaYYYYMMDD & Format(CLng(numeroEncontrado), "0000")
    Else
        If contadorFacturasGlobal = 0 Then contadorFacturasGlobal = 1
        ExtraerNumeroFacturaSimple = "FV-" & fechaYYYYMMDD & Format(contadorFacturasGlobal, "0000")
        contadorFacturasGlobal = contadorFacturasGlobal + 1
    End If
    
    Set regex = Nothing
    Set matches = Nothing
End Function

    Dim fechaParts() As String

    If fechaTexto Like "##/##/####" Or fechaTexto Like "##/##/##" Then

            mes = Format(fechaParts(1), "00")
            anio = fechaParts(2)

Function ConvertirCantidadSimple(cantidad As String) As Double
    If Left(cantidad, 1) = "=" Then
        Dim formula As String
        formula = Mid(cantidad, 2)
        
        Dim parts() As String
        parts = Split(formula, "+")
        Dim total As Double
        total = 0
        
        Dim k As Integer
        For k = 0 To UBound(parts)
            Dim part As String
            part = Trim(parts(k))
            If IsNumeric(part) Then
                total = total + CDbl(part)
            End If
        Next k
        
        ConvertirCantidadSimple = total
    ElseIf IsNumeric(cantidad) Then
        ConvertirCantidadSimple = CDbl(cantidad)
    Else
        ConvertirCantidadSimple = 0
    End If
End Function

Function ObtenerCodigoServicioSimple(descripcion As String) As String
    ' Función para obtener el código de servicio basado en la descripción
    descripcion = LimpiarTexto(descripcion)
    Dim descripcionLower As String
    descripcionLower = LCase(descripcion)
       
    ' 1. PRIMERO: Fotografías con tamaños específicos
    If (InStr(descripcionLower, "fotografía") > 0 Or InStr(descripcionLower, "foto") > 0) Then
        If InStr(descripcionLower, "8x10") > 0 Or InStr(descripcionLower, "8.5x11") > 0 Or _
           InStr(descripcionLower, "8 x 10") > 0 Or InStr(descripcionLower, "8.5 x 11") > 0 Then
            ObtenerCodigoServicioSimple = "F004"
            Exit Function
        ElseIf InStr(descripcionLower, "5x7") > 0 Or InStr(descripcionLower, "5 x 7") > 0 Then
            ObtenerCodigoServicioSimple = "F001"
            Exit Function
        ElseIf InStr(descripcionLower, "1x1") > 0 Or InStr(descripcionLower, "1 x 1") > 0 Then
            ObtenerCodigoServicioSimple = "F002"
            Exit Function
        ElseIf InStr(descripcionLower, "pasaporte") > 0 Or InStr(descripcionLower, "4 fotos") > 0 Then
            ObtenerCodigoServicioSimple = "F003"
            Exit Function
        End If

Sub ExportarFacturaActualFormatoCompacto()
    Dim ws As Worksheet
    Dim libroOrigen As Workbook
    Dim rutaArchivo As String
    Dim contenidoTotal As String
    Dim fso As Object, archivo As Object
    Dim facturasData As Object
    Dim detallesData As Object
    Dim facturaIndex As Long
    Dim detalleIndex As Long
    
    Dim idInicialFact As Long
    Dim idInicialDet As Long
    Dim respuestaIdFact As String
    Dim respuestaIdDet As String
    Dim idActualFact As Long
    Dim idActualDet As Long
    
    Set libroOrigen = ThisWorkbook
    
    On Error Resume Next
    Set ws = ActiveSheet
    On Error GoTo 0
    
    If ws Is Nothing Then
        MsgBox "No hay una hoja activa seleccionada.", vbExclamation
        Exit Sub
    End If
    
    If LCase(ws.Name) = "instrucciones" Or _
       LCase(ws.Name) = "resumen-enero" Or _
       LCase(ws.Name) = "resumen" Or _
       LCase(Left(ws.Name, 7)) = "resumen" Then
        MsgBox "Esta hoja (" & ws.Name & ") no es una factura válida para exportar.", vbExclamation
        Exit Sub
    End If
    
    contadorFacturasGlobal = 1
    contenidoTotal = ""
    facturaIndex = 0
    detalleIndex = 0
    
    Set facturasData = CreateObject("Scripting.Dictionary")
    Set detallesData = CreateObject("Scripting.Dictionary")
    
    rutaArchivo = Application.GetSaveAsFilename( _
        InitialFileName:="Factura_" & ws.Name & "_Compacto.txt", _
        FileFilter:="Archivos de texto (*.txt), *.txt", _
        Title:="Guardar archivo de texto")
    
    If rutaArchivo = "False" Then Exit Sub
    
    ' Preguntar por el ID inicial para tbl_fact
    respuestaIdFact = InputBox("¿En qué número debe comenzar el ID de la tabla 'tbl_fact'?", _
                               "Configuración SQL - Facturas", "1")
                           
    If IsNumeric(respuestaIdFact) And respuestaIdFact <> "" Then
        idInicialFact = CLng(respuestaIdFact)
    Else
        idInicialFact = 1
    End If
    
    ' Preguntar por el ID inicial para tbl_fact_detalle
    respuestaIdDet = InputBox("¿En qué número debe comenzar el ID de la tabla 'tbl_fact_detalle'? (Se incrementará automáticamente para cada detalle)", _
                              "Configuración SQL - Detalles", "1")
                           
    If IsNumeric(respuestaIdDet) And respuestaIdDet <> "" Then
        idInicialDet = CLng(respuestaIdDet)
    Else
        idInicialDet = 1
    End If
    
    idActualFact = idInicialFact
    idActualDet = idInicialDet
    
    Dim idFactUsado As Long
    contenidoTotal = ExtraerContenidoCompacto(ws, facturasData, detallesData, facturaIndex, detalleIndex, _
                                              idActualFact, idActualDet, idFactUsado)
    
    If contenidoTotal = "" Then
        MsgBox "No se encontraron datos válidos en la hoja actual.", vbExclamation
        Exit Sub
    End If
    
    contenidoTotal = contenidoTotal & GenerarEstructuraSQL(facturasData, detallesData, idInicialFact, idInicialDet)
    
    Set fso = CreateObject("Scripting.FileSystemObject")
    Set archivo = fso.CreateTextFile(rutaArchivo, True, True)
    archivo.Write contenidoTotal
    archivo.Close
    
    Shell "notepad.exe " & Chr(34) & rutaArchivo & Chr(34), vbNormalFocus
    
    MsgBox "Exportación completada:" & vbCrLf & _
           "Factura procesada: " & ws.Name & vbCrLf & _
           "ID inicial facturas: " & idInicialFact & vbCrLf & _
           "ID inicial detalles: " & idInicialDet & vbCrLf & _
           "Último ID de detalle usado: " & (idActualDet - 1) & vbCrLf & _
           "Archivo: " & rutaArchivo, vbInformation
    
    Set fso = Nothing
    Set archivo = Nothing
    Set libroOrigen = Nothing
    Set facturasData = Nothing
    Set detallesData = Nothing
End Sub

Function LimpiarTexto(texto As String) As String
    If IsNull(texto) Or texto = "" Then
        LimpiarTexto = ""
        Exit Function
    End If
    
    Dim resultado As String
    resultado = CStr(texto)
    
    resultado = Trim(resultado)
    
    Do While InStr(resultado, "  ") > 0
        resultado = Replace(resultado, "  ", " ")
    Loop
    
    resultado = Replace(resultado, vbLf, " ")
    resultado = Replace(resultado, vbCr, " ")
    resultado = Replace(resultado, Chr(10), " ")
    resultado = Replace(resultado, Chr(13), " ")
    
    Do While InStr(resultado, "  ") > 0
        resultado = Replace(resultado, "  ", " ")
    Loop
    
    LimpiarTexto = Trim(resultado)
End Function
